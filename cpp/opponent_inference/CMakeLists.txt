cmake_minimum_required(VERSION 3.22)
project(opponent_inference LANGUAGES CXX)
set(MODEL_DIR "" CACHE PATH "Directory containing generated.hpp and weights.bin")
set(INFER_MARCH "native" CACHE STRING "Compiler architecture target")
option(INFER_SCALAR "Disable explicit SIMD kernels" OFF)
if(NOT EXISTS "${MODEL_DIR}/generated.hpp")
    message(FATAL_ERROR "Export a model and set -DMODEL_DIR=/absolute/model/path")
endif()
find_package(Threads REQUIRED)
add_executable(opponent_infer_bench bench.cpp)
target_compile_features(opponent_infer_bench PRIVATE cxx_std_20)
target_include_directories(opponent_infer_bench PRIVATE "${MODEL_DIR}")
target_link_libraries(opponent_infer_bench PRIVATE Threads::Threads)
if(INFER_SCALAR)
    target_compile_definitions(opponent_infer_bench PRIVATE NANA_INFER_SCALAR=1)
endif()
if(NOT MSVC)
    target_compile_options(opponent_infer_bench PRIVATE -O3 -march=${INFER_MARCH})
endif()
